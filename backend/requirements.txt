# ---- Web ----
# Exact pins (C103 M20). These were >= floors, which made every deploy install
# whatever was newest that morning — production and a fresh checkout could run
# different code from the same commit, and nothing reviewed the difference. A
# pin moves only in a commit, after the suite has run against it.
#
# Each pin is also at or above the newest fix pip-audit knows of for that
# package (the notes below say which advisory each was raised for), as of
# 2026-10-28; re-check with
#   pip-audit -r <(sed -E 's/\[[a-z,]+\]//' requirements.txt) --no-deps --disable-pip
fastapi==0.162.0
# Transitive, but pinned here because FastAPI's own range admits releases with
# known advisories: starlette below 1.3.1 (7, including multipart DoS) and h11
# below 0.16.0 (request smuggling).
starlette==1.10.0
h11==0.16.0
uvicorn[standard]==0.54.0
python-multipart==0.0.32
email-validator==2.3.0
# Not imported directly: email-validator's DNS dependency. Floor is 2.9.0 for

# -r` keeps whatever is installed for as long as email-validator is satisfied.

# ---- Auth ----
# JWTs are PyJWT's, not python-jose's. Worth stating because `python-jose` was
# the dependency here once, and removing a line from this file does not remove
# the package from a virtualenv that already has it: `pip install -r` installs
# and upgrades, it never uninstalls.
#
# So both python-jose and its `ecdsa` dependency outlived the requirement. As of
# 2026-08-15 the production box still carries an orphaned `ecdsa` 0.19.2 —
# nothing declares it, nothing imports it, and pip-audit reports it every run
# for PYSEC-2026-1325 (a Minerva timing attack on P-256, no fix released).
#
# Nothing loads it, so it is dead weight rather than exposure, but it cannot be
# cleared from here — only on the box, and only by hand:
#
#   sudo -u pulse /opt/Pulse/.venv/bin/pip uninstall -y ecdsa python-jose
#
# Until somebody runs that, an `ecdsa` line in a pip-audit report is this, and
# not a dependency that came back.
PyJWT==2.15.1
bcrypt==5.0.0
# Rate limiting for /auth/* (login guessing, signup, reset-mail flooding).
slowapi==0.1.10
# Imported directly (app.ratelimit: parse_many, the window stats behind
# every 429's wait) rather than only through slowapi, so declared and
# pinned in its own right.
limits==5.8.0
# Credential encryption at rest (Fernet).
# Floor is 50.0 for PYSEC-2026-3552 (a decryption oracle in the PKCS7 helpers).
# Pulse only uses Fernet and never touches those, so the exposure was nil — but
# the floor is what `pip install -r` on the box compares against, and at >=42.0
# an already-installed 49.0.0 satisfied it and was never upgraded.
cryptography==50.0.2
